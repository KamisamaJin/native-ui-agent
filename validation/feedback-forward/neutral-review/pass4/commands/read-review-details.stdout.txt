38: def from_capture(coverage, parent, failed_ids, result_path, result_sha, source_roots, *,
39:                  review_submission=None, review_sha=None, lock=None, lock_sha=None, policy=None, policy_sha=None):

48:     report_path = root / 'automatic.json'

50:         raise ValueError('Original automatic report changed')

59:     report = design.load(report_path)

89:         value['finding_basis'] = 'Recomputed approved-contract review of original state-bound actuals, not an overridden automatic report'
