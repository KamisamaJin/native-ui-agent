 108 def verify_execution(value, plan, lock, lock_sha, fixtures, bridge_sha):
 109     # Reconstructs required after-target extensions from the actual approved plan;
 110     # a caller cannot substitute a sibling base route or drop coverage extensions.
 111     expected = compile_execution(plan, lock, lock_sha, fixtures, bridge_sha,

 113     if expected != value: raise ValueError('Execution manifest differs from independently rebuilt business/observation route')

 117 def _coverage_facts(required, definitions, platform, observation, root=None):
 118     from native_checks import bound_state_evidence, resolve
 119     missing, states = [], []
 120     for entry in required:
 121         parent = {'id': entry['state_id']}
 122         # Exact state definitions are carried in config by the CLI/runner caller
 123         # below; never infer an input epoch from an arbitrary matching label.
 124         parent = next(s for s in definitions if s['id'] == entry['state_id'])
 125         found, complete, reason, capture = bound_state_evidence(observation, parent, root)
 126         target_rows = []
 127         for target in entry['targets']:
 128             seen = []
 129             for index, snapshot in found:
 130                 result = resolve(snapshot, target, platform)
 131                 semantic = result.get('semantic', {})
 132                 if result.get('facts', {}).get('unique') and semantic.get('intersects_viewport'):
 133                     seen.append(index)
 134             target_rows.append({'target': target, 'indices': seen, 'observed': bool(seen)})
 135         good = bool(found) and (complete if entry['coverage'] == 'full_content' else True) and all(v['observed'] for v in target_rows)
 136         if not good: missing.append(entry['state_id'])
 137         states.append({'state_id': entry['state_id'], 'complete': complete, 'satisfied': good, 'reason': reason, 'targets': target_rows,
 138                        **({'coverage_proof': capture['coverage_proof']} if capture and 'coverage_proof' in capture else {})})
 139     return {'required_states': states, 'satisfied': not missing, 'missing_states': missing}

 157 def source_fingerprint(roots):
 158     result = {}
 159     for index, root in enumerate(roots):
 160         root = Path(root).resolve()
 161         if not root.exists(): raise ValueError('Production source root missing')
 162         paths = [root] if root.is_file() else sorted(p for p in root.rglob('*') if p.is_file())
 163         for path in paths:
 164             relative = path.name if root.is_file() else str(path.relative_to(root))
 165             if '__pycache__' in path.parts or path.suffix == '.pyc': continue
 166             result[str(index) + '/' + relative] = hashlib.sha256(path.read_bytes()).hexdigest()
 167     if not result: raise ValueError('Nonempty actual production source set required')
 168     return result

 180 def packet_inventory(root):
 181     root = Path(root).resolve()
 182     files = {}
 183     for path in sorted(root.rglob('*')):
 184         if path == root / 'result.json' or path.name == '.DS_Store' or path.name.startswith('._'):
 185             continue
 186         if path.is_symlink(): raise ValueError('Native packet symlink needs an explicit evidence provider')
 187         if path.is_file(): files[str(path.relative_to(root))] = hashlib.sha256(path.read_bytes()).hexdigest()
