  13 def compile_actions(actions):

  15     result = copy.deepcopy(actions)
  16     for index, action in enumerate(result):
  17         deps = action.get('depends_on', [index - 1] if index else [])
  18         if not isinstance(deps, list) or len(set(deps)) != len(deps) or any(type(i) is not int or not 0 <= i < index for i in deps):
  19             raise ValueError('Action dependencies must be unique earlier action indices')
  20         action['depends_on'] = deps
  21         callbacks = action.get('callback_dependencies', [])
  22         if not isinstance(callbacks, list) or len(callbacks) > 16: raise ValueError('Bounded callback prerequisites required')
  23         for item in callbacks:
  24             if (set(item) != {'action_index', 'expected_events'} or type(item['action_index']) is not int or not 0 <= item['action_index'] < index
  25                     or not isinstance(item['expected_events'], list) or any(type(v) is not str for v in item['expected_events'])):
  26                 raise ValueError('Callback prerequisite needs an earlier action and independently approved event list')
  27         action['callback_dependencies'] = callbacks
  28         conditions = action.get('preconditions', [])
  29         if not isinstance(conditions, list) or len(conditions) > 16: raise ValueError('Bounded native preconditions required')

  61 def replay(actions, steps, platform):
  62     from native_checks import callback_verdict
  63     actions = compile_actions(actions); rows = []
  64     for index, action in enumerate(actions):
  65         blocked = [i for i in action['depends_on'] if i >= len(rows) or not rows[i]['established']]
  66         step = steps[index] if index < len(steps) else {}; proof = step.get('guard', {})
  67         callback_ok = True
  68         for required in action['callback_dependencies']:
  69             previous = required['action_index']
  70             if previous >= len(rows) or not rows[previous]['established'] or events(steps[previous]) != required['expected_events']:
  71                 callback_ok = False; blocked.append(previous)
  72         conditions = proof.get('conditions', [])
  73         condition_ok = (len(conditions) == len(action['preconditions']) and
  74                         all(v.get('condition') == c and condition_passes(c, v.get('actual'))
  75                             for c, v in zip(action['preconditions'], conditions)))
  76         ready = (step.get('action') == action and not blocked and callback_ok and proof.get('protocol') == 1 and proof.get('depends_on') == action['depends_on'] and condition_ok and proof.get('ready') is True)
  77         kind = action['type']
  78         if kind in {'tap', 'apply_next_fixture'}:
  79             delta = step.get('callback_delta')
  80             actual_events = events(step) or []
  81             verdict = callback_verdict(step, actual_events, platform, stimulus='activation' if kind == 'tap' else 'caller_update')
  82             dispatched = verdict['stimulus_established'] and proof.get('binding') == step.get('dispatch', {}).get('binding_before')
  83         else:
  84             dispatched = (step.get('outcome') in {'observed', 'completed'} and bool(step.get('binding_after'))
  85                           and proof.get('binding') == step.get('binding_after'))

  87         # A producer flag cannot certify its own successful activation.
  88         if proof.get('established') is not established: established = False

 107 def verify_observation(actions, observation, platform):
 108     if observation.get('guard_protocol') != 1 or observation.get('guard_platform') != platform or observation.get('guard_actions') != compile_actions(actions):
 109         raise ValueError('Native packet did not execute the exact compiled guarded route')
